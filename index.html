import random import os class Flashcard: def __init__(self, term, definition): self.term = term self.definition = definition self.mastery = 0 # 0: New, 1: Learning, 2: Mastered self.wrong_streak = 0 def clear_screen(): os.system('cls' if os.name == 'nt' else 'clear') def get_multiple_choice_options(current_card, all_cards): # Get the correct answer + 3 random wrong answers options = [current_card.definition] other_defs = [c.definition for c in all_cards if c.definition != current_card.definition] # If there aren't enough cards for 4 choices, just take what's available num_distractors = min(len(other_defs), 3) options.extend(random.sample(other_defs, num_distractors)) random.shuffle(options) return options def study_session(): print("--- Welcome to DIY Quizlet Learn ---") deck = [] # 1. Input Phase print("Enter your cards (leave Term blank to finish):") while True: term = input(f"Term {len(deck)+1}: ").strip() if not term: break definition = input(f"Definition for '{term}': ").strip() deck.append(Flashcard(term, definition)) if len(deck) < 2: print("Add at least 2 cards to start!") return # 2. Study Phase while any(c.mastery < 2 for c in deck): # Pick a card that isn't mastered yet # Prioritize cards with lower mastery or high wrong streaks eligible_cards = [c for c in deck if c.mastery < 2] current_card = random.choice(eligible_cards) clear_screen() print(f"Mastered: {len([c for c in deck if c.mastery == 2])}/{len(deck)}") print(f"Current Target: {current_card.term}\n" + "-"*20) # Decide Mode: 0 mastery = Multiple Choice, 1 mastery = Written if current_card.mastery == 0: print("(Mode: Multiple Choice)") options = get_multiple_choice_options(current_card, deck) for i, opt in enumerate(options): print(f"{i+1}. {opt}") try: choice = int(input("\nYour choice (number): ")) is_correct = options[choice-1] == current_card.definition except (ValueError, IndexError): is_correct = False else: print("(Mode: Written Answer)") ans = input("Type the definition: ").strip().lower() is_correct = ans == current_card.definition.lower() # 3. Update Mastery Logic if is_correct: print("\n✅ CORRECT!") current_card.mastery += 1 current_card.wrong_streak = 0 else: print(f"\n❌ WRONG. The correct answer was: {current_card.definition}") current_card.mastery = 0 # Drop back to New if missed current_card.wrong_streak += 1 input("\nPress Enter to continue...") clear_screen() print("Way to go! You've mastered all cards in this set. 🎉") if __name__ == "__main__": study_session()
